import { test, expect } from '@playwright/test';

test.describe('Lightning Payment Flow', () => {
  test('should create invoice and display QR', async ({ page }) => {
    // Navigate to test page
    await page.goto('/pipeline-test');

    // Verify page loaded
    await expect(page.locator('h1')).toContainText('Pipeline Test');

    // Click run test
    await page.click('[data-run-test]');

    // Wait for invoice creation
    await expect(page.locator('[data-payment-status]')).toContainText('running');

    // Verify QR code appears (invoice created successfully)
    await expect(page.locator('svg[role="img"]')).toBeVisible({ timeout: 10000 });
  });

  test('should complete checkout with payment modal', async ({ page }) => {
    // Navigate to homepage
    await page.goto('/');

    // Add 1-sat product to cart
    await page.click('text=Propaganda Pack >> .. >> button:has-text("Add to Cart")');

    // Go to checkout
    await page.click('text=Proceed to Checkout');

    // Fill form
    await page.fill('input[name="email"]', 'test@example.com');
    await page.fill('input[name="fullName"]', 'Test User');
    await page.fill('input[name="address"]', '21 Satoshi Lane');
    await page.fill('input[name="city"]', 'Bitcoin City');
    await page.fill('input[name="postalCode"]', '21000');
    await page.fill('input[name="country"]', 'Cryptoland');

    // Submit
    await page.click('button:has-text("Pay with Bitcoin")');

    await expect(page.locator('h2:has-text("Lightning Payment")')).toBeVisible();
